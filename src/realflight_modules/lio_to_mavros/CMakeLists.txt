cmake_minimum_required(VERSION 3.0.2)
project(lio_to_mavros)
find_package(catkin REQUIRED COMPONENTS geometry_msgs mavros_msgs nav_msgs std_msgs
  roscpp tf2 tf2_ros tf2_geometry_msgs)
find_package(Eigen3 REQUIRED)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS geometry_msgs mavros_msgs nav_msgs
  std_msgs roscpp tf2 tf2_ros tf2_geometry_msgs DEPENDS EIGEN3)
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
add_executable(lio_to_mavros_node src/lio_to_mavros.cpp)
set_target_properties(lio_to_mavros_node PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON)
add_dependencies(lio_to_mavros_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(lio_to_mavros_node ${catkin_LIBRARIES})
install(TARGETS lio_to_mavros_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(odometry_transform_test test/odometry_transform_test.cpp)
  if(TARGET odometry_transform_test)
    set_target_properties(odometry_transform_test PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON)
  endif()
  find_package(rostest REQUIRED)
  catkin_install_python(PROGRAMS test/test_bridge.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  add_rostest(test/bridge.test DEPENDENCIES lio_to_mavros_node)
endif()
